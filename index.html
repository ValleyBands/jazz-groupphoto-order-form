function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); // Wait up to 10 seconds to prevent concurrent overwrite issues try { var doc = SpreadsheetApp.getActiveSpreadsheet(); var sheet = doc.getActiveSheet(); // Setup headers if sheet is brand new if (sheet.getLastRow() === 0) { sheet.appendRow([ 'Timestamp', 'Order ID', 'Parent Name', 'Email', 'Phone', 'Student Name', 'Jazz Band', '8x10 Formal Qty', '8x10 Fun Qty', '5x7 Sheet Qty', 'Total Student Photos', 'Order Grand Total ($)' ]); sheet.getRange(1, 1, 1, 12).setFontWeight('bold').setBackground('#e0e7ff'); } // Generate a unique Order ID for this single submission var orderId = 'ORD-' + new Date().getTime().toString().slice(-6); var timestamp = new Date(); // Parse incoming data var parentName = e.parameter.parentName || ''; var email = e.parameter.email || ''; var phone = e.parameter.phone || ''; var totalCost = e.parameter.totalCost || '0.00'; var students = JSON.parse(e.parameter.students || '[]'); // Append a row for each student included in the form students.forEach(function(student) { var formalQty = parseInt(student.formalQty) || 0; var funQty = parseInt(student.funQty) || 0; var sheetQty = parseInt(student.sheetQty) || 0; var totalPhotos = parseInt(student.totalPhotos) || (formalQty + funQty + sheetQty); sheet.appendRow([ timestamp, orderId, parentName, email, phone, student.name, student.jazzBand, formalQty, funQty, sheetQty, totalPhotos, '$' + totalCost ]); }); // Return success response for no-cors / Web App execution return ContentService .createTextOutput(JSON.stringify({ 'result': 'success', 'orderId': orderId })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ 'result': 'error', 'error': error.toString() })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
